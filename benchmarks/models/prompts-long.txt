from typing import Dict, List, Optional, Tuple


@dataclass(order=True)
class Entry:
    priority: float
    node: str = field(compare=False)


class Graph:
    """A weighted directed graph stored as adjacency lists."""

    def __init__(self) -> None:
        self.edges: Dict[str, List[Tuple[str, float]]] = {}

    def add_edge(self, src: str, dst: str, weight: float) -> None:
        if weight < 0:
            raise ValueError("negative weights are not supported")
        self.edges.setdefault(src, []).append((dst, weight))
        self.edges.setdefault(dst, [])

    def neighbors(self, node: str) -> List[Tuple[str, float]]:
        return self.edges.get(node, [])

    def shortest_paths(self, source: str) -> Tuple[Dict[str, float], Dict[str, Optional[str]]]:
        """Dijkstra's algorithm: the distance to every node and the predecessor on a shortest path."""
        dist: Dict[str, float] = {node: float("inf") for node in self.edges}
        prev: Dict[str, Optional[str]] = {node: None for node in self.edges}
        dist[source] = 0.0
        queue: List[Entry] = [Entry(0.0, source)]
        while queue:
            entry = heapq.heappop(queue)
            if entry.priority > dist[entry.node]:
                continue
            for nxt, weight in self.neighbors(entry.node):
                candidate = dist[entry.node] + weight
                if candidate < dist[nxt]:
                    dist[nxt] = candidate
                    prev[nxt] = entry.node
                    heapq.heappush(queue, Entry(candidate, nxt))
        return dist, prev

    def path(self, source: str, target: str) -> List[str]:
        """The nodes of a shortest path from source to target, or an empty list when target is unreachable."""
        dist, prev = self.shortest_paths(source)
        if dist.get(target, float("inf")) == float("inf"):
            return []
        nodes = []
        at: Optional[str] = target
        while at is not None:
            nodes.append(at)
            at = prev[at]
        return list(reversed(nodes))
