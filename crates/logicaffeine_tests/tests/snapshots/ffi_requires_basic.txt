#[allow(unused_imports)]
use std::fmt::Write as _;
use logicaffeine_data::*;
use logicaffeine_system::*;

fn main() {
    std::panic::set_hook(::std::boxed::Box::new(|_logos_info| {
        logicaffeine_system::io::flush_stdout();
        let _logos_msg = _logos_info.payload().downcast_ref::<&str>().map(|s| s.to_string())
            .or_else(|| _logos_info.payload().downcast_ref::<String>().cloned())
            .unwrap_or_else(|| "LOGOS runtime error".to_string());
        eprintln!("{}", _logos_msg);
        if !logicaffeine_system::failure::is_caught() { std::process::exit(1); }
    }));
    logicaffeine_system::stack::run_main(_logos_main);
}
fn _logos_main() {
    let x: i64 = 42;
    show(&42);
}
